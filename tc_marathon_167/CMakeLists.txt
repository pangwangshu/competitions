cmake_minimum_required(VERSION 3.16)
project(mm167 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()

# src/WordGame.cpp is a unity translation unit that includes every module, so
# the solver and the submission bundle (scripts/bundle.py) compile the same code.
add_executable(WordGame src/WordGame.cpp)
set(TESTS engine_test simulation_test deep_test discard_test fishing_test)
foreach(t ${TESTS})
  add_executable(${t} tests/${t}.cpp)
endforeach()
foreach(target WordGame ${TESTS})
  target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
endforeach()

option(SANITIZE "Enable address and undefined behavior sanitizers" OFF)
if(SANITIZE)
  foreach(target WordGame ${TESTS})
    target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(${target} PRIVATE -fsanitize=address,undefined)
  endforeach()
endif()

enable_testing()
add_test(NAME engine COMMAND engine_test)
add_test(NAME simulation COMMAND simulation_test)
add_test(NAME deep COMMAND deep_test)
add_test(NAME discard COMMAND discard_test)
add_test(NAME fishing COMMAND fishing_test)
