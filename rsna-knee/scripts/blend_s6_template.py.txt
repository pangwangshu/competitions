"""Blend submission without OAI: nartaa's S6 ConvNeXt-small (public 0.950 alone, no OAI, CC BY-NC 4.0) and our reader,
mixed per finding in rank space.

Generated by scripts/build_blend_s6_kernel.py. The public script nartaa/rsna-knee-s6fall-solo is imported as a module (its
component A source blanked) and only its component B is called, through its own run_b, on the test studies; its own retry
and failure cap apply (hidden set: at most max(3, 0.25 % of N) studies may stay unread; 3-study commit: none). A study S6
cannot read takes our reader's rank. On the 3-study commit run any failure raises; on the hidden set a failed side falls
back to the other one.
"""
K1_SOURCE = __K1_SOURCE__
import importlib.util
import json
import os
import time
from pathlib import Path

import pandas as pd

from rsna_knee.evaluation import find_one, sha256, submission
from rsna_knee.reader.paths import DATA

config = json.loads(os.environ["RK_CONFIG_JSON"])
w = float(config["blend"]["s6_weight"])
budget = float(config["blend"].get("s6_budget_s", 4 * 3600))
out = Path("/kaggle/working")
sample = pd.read_csv(Path(DATA) / "sample_submission.csv", dtype={"StudyInstanceUID": str})
labels = [c for c in sample.columns if c != "StudyInstanceUID"]
strict = len(sample) <= 3
t0, report = time.time(), {"s6_weight": w, "strict": strict, "studies": len(sample)}


def ranks(frame):
    frame = frame.set_index("StudyInstanceUID").loc[sample.StudyInstanceUID, labels]
    if not frame.notna().all().all():
        raise RuntimeError("Missing predictions")
    return frame.rank(pct=True)


def side(name, run):
    start = time.time()
    try:
        result = run()
        report[name] = {"ok": True, "seconds": time.time() - start, **report.get(name, {})}
        return result
    except Exception as error:
        report[name] = {"ok": False, "seconds": time.time() - start, "error": f"{type(error).__name__}: {error}"[:3000]}
        print(f"[blend] {name} FAILED: {report[name]['error']}", flush=True)
        if strict:
            raise
        return None


def s6():
    """(ranks with failed rows filled by the column mean, failed study ids)."""
    src = Path("/tmp/rk_k1.py")
    src.write_text(K1_SOURCE)
    spec = importlib.util.spec_from_file_location("rk_k1", src)
    k1 = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(k1)
    assert k1.MODE == "submit"
    k1.OUT.mkdir(parents=True, exist_ok=True)
    k1.TMP.mkdir(parents=True, exist_ok=True)
    k1._LOG = open(k1.OUT / "k1_log.txt", "a")
    k1._RECEIPT["env"] = k1.probe_env()
    if not k1._RECEIPT["env"].get("cuda_available"):
        raise RuntimeError("CUDA is not available")
    I = k1.prepare_inputs()
    if sorted(I.test_ids) != sorted(sample.StudyInstanceUID):
        raise RuntimeError("S6 test ids differ from sample_submission")
    print("B probe", json.dumps(k1.probe_b_stack(I)), flush=True)
    try:
        P, _, info, failed = k1.run_b(I, "B", I.test_ids, "test", budget)
    finally:
        k1.kill_all_children()
    report["s6"] = {"failed": failed, "parts": info["parts"], "retry": None if info["retry"] is None else len(info["retry"]["ids"])}
    frame = pd.DataFrame(P, columns=k1.LAB)
    frame.insert(0, "StudyInstanceUID", I.test_ids)
    frame.to_csv(out / "s6_probs.csv", index=False, float_format="%.7g")
    frame = frame.set_index("StudyInstanceUID").loc[sample.StudyInstanceUID, labels]
    bad = frame.isna().any(axis=1)
    if bad.all():
        raise RuntimeError("S6 scored no study")
    frame = frame.fillna(frame[~bad].mean())         # as nartaa's blend_with_failures; these rows are replaced below
    return frame.rank(pct=True), list(frame.index[bad])


def ours():
    paths = []
    for c in config["ckpts"]:
        hits = find_one(c["file"])
        if len({sha256(h) for h in hits}) != 1:
            raise ValueError(f"Ambiguous checkpoint {c['file']}: {hits}")
        paths.append(hits[0])
    reader = {k: v for k, v in config.items() if k != "blend"}
    submission(reader, data=DATA, ckpt_paths=paths, out=out / "own")
    return ranks(pd.read_csv(out / "own" / "submission.csv", dtype={"StudyInstanceUID": str}))


got = side("s6", s6)                                  # S6 first: its GPU worker processes exit before our reader loads
r_own = side("own", ours)
r_s6, failed = got if got is not None else (None, [])
if r_s6 is not None and r_own is not None:
    blend = w * r_s6 + (1 - w) * r_own
    blend.loc[failed] = r_own.loc[failed]             # a study S6 could not read takes our rank
    blend = blend.rank(pct=True)
elif r_s6 is not None:
    if failed:                                       # no reader rank to give them: they keep the column-mean rank
        report["s6_failed_without_own"] = len(failed)
    blend = r_s6
elif r_own is not None:
    blend = r_own
else:
    raise RuntimeError("Both S6 and our reader failed")
report["used"] = "blend" if r_s6 is not None and r_own is not None else ("s6" if r_s6 is not None else "own")
report["seconds"] = time.time() - t0
sub = blend.reset_index()[list(sample.columns)]
assert sub.StudyInstanceUID.tolist() == sample.StudyInstanceUID.tolist() and sub[labels].notna().all().all()
sub.to_csv(out / "submission.csv", index=False)
json.dump(report, open(out / "blend_run.json", "w"), indent=1)
print("[blend]", json.dumps(report), flush=True)
