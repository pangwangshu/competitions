"""Teacher probabilities for S5 T4: nartaa's public CoAtNet (LB 0.949) scored on training studies.

Generated by scripts/build_teacher_kernel.py; the model, preprocessing and TTA are the public
notebook nartaa/rsna-knee-0949-anatomical-mirror, unchanged, run chunk by chunk in subprocesses.
"""
INNER_SOURCE = __INNER_SOURCE__
import hashlib
import json
import os
import subprocess
import sys
import time
from pathlib import Path

import pandas as pd

config = json.loads(os.environ.get("RK_CONFIG_JSON", "{}"))
real = next(p for p in ("/kaggle/input/competitions/rsna-knee-abnormality-detection",
                        "/kaggle/input/rsna-knee-abnormality-detection") if os.path.exists(p + "/train.csv"))
train = pd.read_csv(real + "/train.csv", dtype={"StudyInstanceUID": str})
labels = [c for c in train.columns if c not in ("StudyInstanceUID", "Report")]
ids = train.StudyInstanceUID
if config.get("studies") == "gold":
    ids = ids[train[labels].notna().all(axis=1)]
ids = sorted(ids)[: config.get("limit") or None]
weights = list(Path("/kaggle/input").rglob("raptor_ft_alldata_t16_blendjev_oai_d96_r384_swa.pt"))
assert weights, "teacher weights are not attached"
with weights[0].open("rb") as f:
    assert hashlib.file_digest(f, "sha256").hexdigest() == "__WEIGHT_SHA__", "Checkpoint hash differs"
series = pd.read_csv(real + "/train_series.csv", dtype=str)
inner = Path("/tmp/rk_teacher_inner.py")
inner.write_text(INNER_SOURCE)
out = Path("/kaggle/working")
size = int(config.get("chunk", 500))
t0, parts, report = time.time(), [], []
print("teacher studies:", len(ids), "| chunks of", size, flush=True)
for n, start in enumerate(range(0, len(ids), size)):
    chunk = ids[start:start + size]
    root = Path(f"/tmp/rk_teacher_root_{n}")
    root.mkdir(parents=True, exist_ok=True)
    pd.DataFrame({"StudyInstanceUID": chunk}).to_csv(root / "test.csv", index=False)
    series[series.StudyInstanceUID.isin(set(chunk))].to_csv(root / "test_series.csv", index=False)
    if not (root / "test_series").exists():
        os.symlink(real + "/train_series", root / "test_series")
    part = out / f"teacher_probs_part{n:02d}.csv"
    env = dict(os.environ, RK_TEACHER_ROOT=str(root), RK_TEACHER_OUT=str(part))
    code = subprocess.run([sys.executable, str(inner)], env=env, cwd="/tmp").returncode
    ok = code == 0 and part.exists() and len(pd.read_csv(part)) == len(chunk)
    report.append({"chunk": n, "studies": len(chunk), "return_code": code, "ok": bool(ok),
                   "elapsed": time.time() - t0})
    print("chunk", n, report[-1], flush=True)
    json.dump(report, open(out / "teacher_run.json", "w"), indent=1)
    if ok:
        parts.append(pd.read_csv(part, dtype={"StudyInstanceUID": str}))
if parts:
    pd.concat(parts).to_csv(out / "teacher_probs.csv", index=False, float_format="%.7g")
if not all(r["ok"] for r in report):
    raise RuntimeError(f"Teacher chunks failed: {[r['chunk'] for r in report if not r['ok']]}")
print("DONE", len(ids), "studies in", round(time.time() - t0), "s", flush=True)
