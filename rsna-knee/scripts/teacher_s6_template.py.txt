"""T4 teacher (S5-T4S6): nartaa's S6 ConvNeXt-small (public 0.950 alone, no OAI, gold-58 excluded) on training studies.

Generated by scripts/build_teacher_s6_kernel.py. The public script nartaa/rsna-knee-s6fall-solo is imported as a module and
only its component B (Heliosli's compact_infer.py with the SKM-TEA anatomy V-Net, model E_prior_s42_fall.pt) is called,
through its own run_b, on chunks of training studies. Component A (the OAI-trained CoAtNet) is never run, and its embedded
source is blanked by the builder, so a study B cannot read stays missing instead of taking the CoAtNet's score.
"""
K1_SOURCE = __K1_SOURCE__
import importlib.util
import json
import math
import os
import shutil
import time
from pathlib import Path

import pandas as pd

from rsna_knee.reader.make_labels import report_group, stable_group_folds

config = json.loads(os.environ.get("RK_CONFIG_JSON", "{}"))
work = Path("/kaggle/working")
work.mkdir(parents=True, exist_ok=True)
src = Path("/tmp/rk_k1.py")
src.write_text(K1_SOURCE)
spec = importlib.util.spec_from_file_location("rk_k1", src)
k1 = importlib.util.module_from_spec(spec)
spec.loader.exec_module(k1)
k1.MODE = "parity"                                    # prepare_inputs then also reads train_series.csv (per-study cost)
k1.b_fail_cap = lambda n: max(2, math.ceil(0.01 * n))  # studies still failing after the retry stay missing, up to 1 %
k1.TMP.mkdir(parents=True, exist_ok=True)
k1._LOG = open(work / "k1_log.txt", "a")
k1._RECEIPT["env"] = k1.probe_env()
if not k1._RECEIPT["env"].get("cuda_available"):
    raise RuntimeError("CUDA is not available")
I = k1.prepare_inputs()
print("B probe", json.dumps(k1.probe_b_stack(I)), flush=True)

train = pd.read_csv(I.comp / "train.csv", dtype={"StudyInstanceUID": str})
labels = [c for c in train.columns if c not in ("StudyInstanceUID", "Report")]
gold = train[labels].notna().all(axis=1)
fold = pd.Series(-1, index=train.index)
fold[~gold] = stable_group_folds(train.Report[~gold].map(report_group).values)  # = labels_v0 folds (checked locally)
studies = config.get("studies", "gold")
keep = {"gold": gold, "fold0_train": gold | (fold > 0), "all": gold | ~gold}[studies]
ids = sorted(train.StudyInstanceUID[keep])[: config.get("limit") or None]
size, budget = int(config.get("chunk", 500)), float(config.get("chunk_budget_s", 4 * 3600))
print(f"teacher studies: {len(ids)} ({studies}) | chunks of {size} | budget {budget:.0f}s per chunk", flush=True)
t0, parts, report = time.time(), [], []
for n, start in enumerate(range(0, len(ids), size)):
    chunk = ids[start:start + size]
    k1.OUT = work / f"chunk{n:02d}"
    t = time.time()
    entry = {"chunk": n, "studies": len(chunk)}
    try:
        P, _, info, failed = k1.run_b(I, "B", chunk, "train", budget)
        df = pd.DataFrame(P, columns=k1.LAB)
        df.insert(0, "StudyInstanceUID", chunk)
        parts.append(df.dropna())
        entry.update(ok=True, failed=len(failed), failed_ids=failed, seconds=time.time() - t,
                     parts=info["parts"], retry=None if info["retry"] is None else len(info["retry"]["ids"]))
    except Exception as error:                        # one bad chunk costs that chunk, not the run
        k1.kill_all_children()
        entry.update(ok=False, failed=len(chunk), seconds=time.time() - t, error=repr(error)[:3000])
    for tmp in k1.TMP.glob("B_tmp*"):
        shutil.rmtree(tmp, ignore_errors=True)
    report.append(entry)
    print("chunk", n, json.dumps({k: v for k, v in entry.items() if k != "failed_ids"}), flush=True)
    json.dump(report, open(work / "teacher_run.json", "w"), indent=1)
    if parts:
        pd.concat(parts).to_csv(work / "teacher_probs_s6.csv", index=False, float_format="%.7g")
done = sum(len(p) for p in parts)
print("DONE", done, "of", len(ids), "studies in", round(time.time() - t0), "s", flush=True)
if done == 0:
    raise RuntimeError("The teacher scored no study")
