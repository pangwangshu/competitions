"""Blend submission: nartaa's public CoAtNet (LB 0.949) and our reader, mixed per finding in rank space.

Generated by scripts/build_blend_kernel.py; the CoAtNet is the public notebook
nartaa/rsna-knee-0949-anatomical-mirror, unchanged except for its output path, run in a subprocess.
"""
COATNET_SOURCE = __COATNET_SOURCE__
import hashlib
import json
import os
import subprocess
import sys
import time
from pathlib import Path

import pandas as pd

from rsna_knee.evaluation import find_one, sha256, submission
from rsna_knee.reader.paths import DATA

config = json.loads(os.environ["RK_CONFIG_JSON"])
w = float(config["blend"]["coatnet_weight"])
out = Path("/kaggle/working")
sample = pd.read_csv(Path(DATA) / "sample_submission.csv", dtype={"StudyInstanceUID": str})
labels = [c for c in sample.columns if c != "StudyInstanceUID"]
# The commit run sees the 3 visible studies: there every failure raises, so a fallback cannot pass as a
# blend. On the hidden set a failed side falls back to the other one.
strict = len(sample) <= 3
t0, report = time.time(), {"coatnet_weight": w, "strict": strict, "studies": len(sample)}


def ranks(frame):
    frame = frame.set_index("StudyInstanceUID").loc[sample.StudyInstanceUID, labels]
    if not frame.notna().all().all():
        raise RuntimeError("Missing predictions")
    return frame.rank(pct=True)


def side(name, run):
    start = time.time()
    try:
        result = ranks(run())
        report[name] = {"ok": True, "seconds": time.time() - start}
        return result
    except Exception as error:
        report[name] = {"ok": False, "seconds": time.time() - start, "error": f"{type(error).__name__}: {error}"}
        print(f"[blend] {name} FAILED: {report[name]['error']}", flush=True)
        if strict:
            raise
        return None


def coatnet():
    weights = find_one("raptor_ft_alldata_t16_blendjev_oai_d96_r384_swa.pt")  # depth-limited: never walk the DICOMs
    with open(weights[0], "rb") as f:
        assert hashlib.file_digest(f, "sha256").hexdigest() == "__WEIGHT_SHA__", "Checkpoint hash differs"
    script, result = Path("/tmp/rk_coatnet.py"), out / "coatnet_submission.csv"
    script.write_text(COATNET_SOURCE)
    env = dict(os.environ, RK_COATNET_OUT=str(result))
    code = subprocess.run([sys.executable, str(script)], env=env, cwd="/tmp").returncode
    if code != 0:
        raise RuntimeError(f"CoAtNet exited with {code}")
    return pd.read_csv(result, dtype={"StudyInstanceUID": str})


def ours():
    paths = []
    for c in config["ckpts"]:
        hits = find_one(c["file"])
        if len({sha256(h) for h in hits}) != 1:
            raise ValueError(f"Ambiguous checkpoint {c['file']}: {hits}")
        paths.append(hits[0])
    reader = {k: v for k, v in config.items() if k != "blend"}
    submission(reader, data=DATA, ckpt_paths=paths, out=out / "own")
    return pd.read_csv(out / "own" / "submission.csv", dtype={"StudyInstanceUID": str})


r_coat = side("coatnet", coatnet)
r_own = side("own", ours)
if r_coat is not None and r_own is not None:
    blend = (w * r_coat + (1 - w) * r_own).rank(pct=True)
elif r_coat is not None or r_own is not None:
    blend = r_coat if r_coat is not None else r_own
else:
    raise RuntimeError("Both CoAtNet and our reader failed")
report["used"] = "blend" if r_coat is not None and r_own is not None else ("coatnet" if r_coat is not None else "own")
report["seconds"] = time.time() - t0
sub = blend.reset_index()[list(sample.columns)]
assert sub.StudyInstanceUID.tolist() == sample.StudyInstanceUID.tolist() and sub[labels].notna().all().all()
sub.to_csv(out / "submission.csv", index=False)
json.dump(report, open(out / "blend_run.json", "w"), indent=1)
print("[blend]", json.dumps(report), flush=True)
